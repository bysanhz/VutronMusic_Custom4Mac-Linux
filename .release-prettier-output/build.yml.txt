name: Release

on:
  push:
    tags:
      - 'v*.*.*'
  workflow_dispatch: {}

permissions:
  contents: write

concurrency:
  group: release-${{ github.ref }}
  cancel-in-progress: false

jobs:
  verify:
    name: Verify release source
    runs-on: ubuntu-latest

    steps:
      - name: Checkout source
        uses: actions/checkout@v6

      - name: Setup Node.js
        uses: actions/setup-node@v6
        with:
          node-version: '22'
          cache: yarn

      - name: Verify tag matches package version
        if: github.event_name == 'push'
        shell: bash
        run: |
          set -euo pipefail
          PACKAGE_VERSION="$(node -p "require('./package.json').version")"
          TAG_VERSION="${GITHUB_REF_NAME#v}"
          echo "package.json version: ${PACKAGE_VERSION}"
          echo "tag version: ${TAG_VERSION}"
          test "${PACKAGE_VERSION}" = "${TAG_VERSION}"

      - name: Install dependencies without native rebuild
        run: yarn install --frozen-lockfile --ignore-scripts --network-timeout 600000

      - name: Check formatting
        run: yarn run format:check

      - name: Lint source
        run: yarn lint

      - name: Type check
        run: yarn run vue:type-check

      - name: Run security and feature regression tests
        run: >-
          yarn playwright test tests/security-hardening.spec.ts tests/feature-regression.spec.ts


      - name: Build renderer and main process
        run: yarn vite --config vite.config.ts build

  Release:
    name: Build ${{ matrix.platform }} ${{ matrix.arch }}
    needs: verify
    strategy:
      fail-fast: false
      matrix:
        include:
          - os: macos-15-intel
            arch: x64
            platform: darwin
          - os: macos-14
            arch: arm64
            platform: darwin
          - os: windows-2022
            arch: x64
            platform: win32
          - os: windows-11-arm
            arch: arm64
            platform: win32
          - os: ubuntu-22.04
            arch: x64
            platform: linux
          - os: ubuntu-22.04-arm
            arch: arm64
            platform: linux

    runs-on: ${{ matrix.os }}
    env:
      XVFB_PFX: ${{ (matrix.platform == 'linux' && 'xvfb-run --auto-servernum -- ') || '' }}

    steps:
      - name: Validate architecture match
        run: |
          echo "Runner OS: ${{ matrix.os }}"
          echo "Target architecture: ${{ matrix.arch }}"
          echo "Architecture validation passed"

      - name: Checkout source
        uses: actions/checkout@v6
        with:
          submodules: recursive

      - name: Setup Node.js
        uses: actions/setup-node@v6
        with:
          node-version: '22'
          cache: yarn

      - name: Install build essentials on macOS
        if: matrix.platform == 'darwin'
        run: brew install rpm

      - name: Install build essentials on Linux x64
        if: matrix.platform == 'linux' && matrix.arch == 'x64'
        run: |
          sudo apt-get update
          sudo apt-get install -y rpm fakeroot snapd libnss3-dev libgtk-3-dev libxss-dev libgconf-2-4 libasound2
          sudo snap install snapcraft --classic

      - name: Install build essentials on Linux ARM64
        if: matrix.platform == 'linux' && matrix.arch == 'arm64'
        run: |
          sudo apt-get update
          sudo apt-get install -y ruby ruby-dev build-essential rpm fakeroot libnss3 libgtk-3-0 libxss1 libgconf-2-4 libasound2
          sudo gem install fpm
          rm -rf ~/.cache/electron-builder/fpm || true
          fpm --version

      - name: Install dependencies
        env:
          SKIP_REBUILD: true
        run: ${{ env.XVFB_PFX }}yarn install --frozen-lockfile --network-timeout 600000

      - name: Fix taglib-wasm
        run: ${{ env.XVFB_PFX }}npm run fix-taglib

      - name: Fix app-builder permissions on Linux ARM64
        if: matrix.platform == 'linux' && matrix.arch == 'arm64'
        run: chmod +x node_modules/app-builder-bin/linux/arm64/app-builder || true

      - name: Rebuild native dependencies
        run: |
          ${{ env.XVFB_PFX }}npm rebuild
          ${{ env.XVFB_PFX }}./node_modules/.bin/electron-rebuild

      - name: Build Linux ARM64 packages
        if: matrix.platform == 'linux' && matrix.arch == 'arm64'
        shell: bash
        env:
          GH_TOKEN: ${{ github.token }}
          ARCH: ${{ matrix.arch }}
          USE_SYSTEM_FPM: 'true'
        run: yarn build --${{ matrix.arch }} -p never --linux deb --linux rpm --linux AppImage

      - name: Build Linux x64 packages
        if: matrix.platform == 'linux' && matrix.arch == 'x64'
        shell: bash
        env:
          GH_TOKEN: ${{ github.token }}
          ARCH: ${{ matrix.arch }}
        run: yarn build --${{ matrix.arch }} -p never --linux deb --linux rpm --linux AppImage --linux snap

      - name: Build macOS or Windows packages
        if: matrix.platform != 'linux'
        shell: bash
        env:
          GH_TOKEN: ${{ github.token }}
          ARCH: ${{ matrix.arch }}
          CSC_IDENTITY_AUTO_DISCOVERY: 'false'
        run: yarn build --${{ matrix.arch }} -p never

      - name: Upload Windows installer
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'win32'
        with:
          name: VutronMusic-win-${{ matrix.arch }}-installer
          path: release/*/VutronMusic*_${{ matrix.arch }}.exe
          if-no-files-found: ignore

      - name: Upload Windows portable package
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'win32'
        with:
          name: VutronMusic-win-${{ matrix.arch }}-portable
          path: release/*/VutronMusic*_${{ matrix.arch }}-Portable.exe
          if-no-files-found: ignore

      - name: Upload Windows metadata
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'win32'
        with:
          name: VutronMusic-win-${{ matrix.arch }}-metadata
          path: |
            release/*/VutronMusic*.blockmap
            release/*/latest*.yml
          if-no-files-found: ignore

      - name: Upload macOS DMG
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'darwin'
        with:
          name: VutronMusic-mac-${{ matrix.arch }}-dmg
          path: release/*/VutronMusic*_${{ matrix.arch }}.dmg
          if-no-files-found: error

      - name: Upload macOS metadata
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'darwin'
        with:
          name: VutronMusic-mac-${{ matrix.arch }}-metadata
          path: |
            release/*/VutronMusic*.blockmap
            release/*/latest*.yml
          if-no-files-found: ignore

      - name: Upload Linux AppImage
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'linux'
        with:
          name: VutronMusic-linux-${{ matrix.arch }}-AppImage
          path: release/*/VutronMusic*.AppImage
          if-no-files-found: error

      - name: Upload Linux Deb
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'linux'
        with:
          name: VutronMusic-linux-${{ matrix.arch }}-deb
          path: release/*/VutronMusic*.deb
          if-no-files-found: error

      - name: Upload Linux RPM
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'linux'
        with:
          name: VutronMusic-linux-${{ matrix.arch }}-rpm
          path: release/*/VutronMusic*.rpm
          if-no-files-found: ignore

      - name: Upload Linux Snap
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'linux' && matrix.arch == 'x64'
        with:
          name: VutronMusic-linux-${{ matrix.arch }}-snap
          path: release/*/VutronMusic*.snap
          if-no-files-found: ignore

      - name: Upload Linux metadata
        uses: actions/upload-artifact@v6
        if: matrix.platform == 'linux'
        with:
          name: VutronMusic-linux-${{ matrix.arch }}-metadata
          path: |
            release/*/VutronMusic*.blockmap
            release/*/latest*.yml
          if-no-files-found: ignore

  publish-release:
    name: Publish GitHub Release
    needs: Release
    runs-on: ubuntu-22.04

    steps:
      - name: Checkout release metadata
        uses: actions/checkout@v6

      - name: Download all artifacts
        uses: actions/download-artifact@v6
        with:
          path: all-artifacts

      - name: Flatten release assets
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p upload
          while IFS= read -r -d '' file; do
            base="$(basename "$file")"
            dest="upload/$base"
            if [[ "$base" == latest*.yml && -f "$dest" ]]; then
              old_size="$(stat -c%s "$dest")"
              new_size="$(stat -c%s "$file")"
              if (( new_size > old_size )); then
                cp "$file" "$dest"
              fi
            else
              cp "$file" "$dest"
            fi
          done < <(find all-artifacts -type f -print0)
          find upload -maxdepth 1 -type f -print

      - name: Resolve release tag
        shell: bash
        run: |
          if [[ "${GITHUB_EVENT_NAME}" == "workflow_dispatch" ]]; then
            echo "RELEASE_TAG=v-draft-$(date +'%Y%m%d%H%M')" >> "$GITHUB_ENV"
            echo "RELEASE_IS_DRAFT=true" >> "$GITHUB_ENV"
          else
            echo "RELEASE_TAG=${GITHUB_REF_NAME}" >> "$GITHUB_ENV"
            echo "RELEASE_IS_DRAFT=false" >> "$GITHUB_ENV"
          fi

      - name: Verify formal release notes
        if: github.event_name == 'push'
        shell: bash
        run: test -f ".github/release-notes/${RELEASE_TAG}.md"

      - name: Upload and publish formal release
        if: github.event_name == 'push'
        uses: softprops/action-gh-release@v3
        with:
          files: upload/*
          body_path: .github/release-notes/${{ env.RELEASE_TAG }}.md
          draft: false
          prerelease: false
          generate_release_notes: false
          tag_name: ${{ env.RELEASE_TAG }}
          fail_on_unmatched_files: true

      - name: Upload draft release
        if: github.event_name == 'workflow_dispatch'
        uses: softprops/action-gh-release@v3
        with:
          files: upload/*
          draft: true
          prerelease: false
          generate_release_notes: true
          tag_name: ${{ env.RELEASE_TAG }}
          fail_on_unmatched_files: true
